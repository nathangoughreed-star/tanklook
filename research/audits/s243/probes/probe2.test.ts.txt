it('counts', async () => {

    noCare: c(s => !CARE[s.id]),
    temp: c(s => !!CARE[s.id]?.temp, wet), noTemp: c(s => !CARE[s.id]?.temp, wet),

    noTankMinLine: c(s => !minDataOf(s.id)?.red, wet),
    noTemper: c(s => CARE[s.id]?.temper === undefined, wet),
    withPrey: c(s => !!CARE[s.id]?.prey, wet),
    withTerritory: c(s => !!(CARE[s.id] as any)?.territory || !!(CARE[s.id] as any)?.males || !!CARE[s.id]?.solo || (CARE[s.id] as any)?.max !== undefined, wet),
    noMass: c(s => !(bm.species as any)[s.id], wet), massEstimate: c(s => (bm.species as any)[s.id]?.src === 'estimate', wet),
    ownSpaceFactor: c(s => !!(SPACE_FACTOR as any)[s.id], wet),
    withNeeds: 0,
  };
  (await import('node:fs')).writeFileSync('C:/Users/nreed/AppData/Local/Temp/claude/C--Users-nreed-Glass-Box--claude-worktrees-peaceful-hertz-ad3d69/d0745941-beaf-4110-b9ea-0e33b2ed8f70/scratchpad/counts.json', JSON.stringify(out, null, 1));
});
