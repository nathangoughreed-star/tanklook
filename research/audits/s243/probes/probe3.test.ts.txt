const STD: [number, number[]][] = [[5, [16, 8, 10]], [10, [20, 10, 12]], [15, [24, 12, 12]], [20, [24, 12, 16]], [29, [30, 12, 18]], [30, [36, 12, 16]], [40, [36, 18, 16]], [55, [48, 13, 21]], [75, [48, 18, 21]], [90, [48, 18, 24]], [125, [72, 18, 21]]];

it('sweep', async () => {
  const out: string[] = []; let filtBind = 0, over = 0; const bySrc: Record<string, number[]> = { room: [], temper: [] };

    const r = stockingOf(s); if (r.filtrationPercent > r.spacePercent) filtBind++; (bySrc[l.cap] ??= []).push(r.spacePercent);
    out.push(`${l.cap}\t${l.name}\tspace ${r.spacePercent.toFixed(0)}\tfilt ${r.filtrationPercent.toFixed(0)}`);

    const b = base(sc); Object.assign(SPACE_NUDGE, ns); Object.assign(GROUP_GROW, gg);
    out.push(`${sc.name}: as shown ${a.spacePercent.toFixed(0)} (water ${a.waterPercent.toFixed(0)}, floor ${a.floorPercent.toFixed(0)}) ; without hand-set nudge/grow ${b.spacePercent.toFixed(0)} (water ${b.waterPercent.toFixed(0)}, floor ${b.floorPercent.toFixed(0)})`);
  }
  (await import('node:fs')).writeFileSync('C:/Users/nreed/AppData/Local/Temp/claude/C--Users-nreed-Glass-Box--claude-worktrees-peaceful-hertz-ad3d69/d0745941-beaf-4110-b9ea-0e33b2ed8f70/scratchpad/sweep.txt', out.join('\n'));
});
